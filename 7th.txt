
N = 8

board = [[0 for i in range(N)] for j in range(N)]

steps = []


def display(b):
    print("  1 2 3 4 5 6 7 8")

    for i in range(N):
        print(i + 1, end=" ")

        for j in range(N):
            if b[i][j] == 1:
                print("Q", end=" ")
            else:
                print(".", end=" ")

        print()

    print("------------------------")


def save_step(message):
    steps.append((message, [row[:] for row in board]))


def is_safe(row, col):

    for i in range(row):
        if board[i][col] == 1:
            return False

    i = row - 1
    j = col - 1

    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    i = row - 1
    j = col + 1

    while i >= 0 and j < N:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


def solve(row):

    if row == N:
        return True

    for col in range(N):

        if is_safe(row, col):

            board[row][col] = 1

            save_step("Place Queen at Row " + str(row + 1) +
                      ", Column " + str(col + 1))

            if solve(row + 1):
                return True

            board[row][col] = 0

            save_step("Remove Queen from Row " + str(row + 1) +
                      ", Column " + str(col + 1))

    return False


print("INITIAL CHESSBOARD")
display(board)

if solve(0):

    for i, (message, state) in enumerate(steps, 1):
        print("\nStep", i, ":", message)
        display(state)

    print("\nFINAL SOLUTION")
    display(board)

    print("All 8 Queens placed successfully.")
    print("Total Steps:", len(steps))

else:
    print("No Solution Found")
